# 1. THE HARDWARE CONFIG
# Edit these values to match your specific printer setup
[gcode_macro V_DAR_CONFIG]
variable_nozzle_size: 0.6    # Diameter of your nozzle (e.g. 0.4, 0.6, 0.8)
variable_scan_speed: 300     # Speed of the laser sweep (F300 = 5mm/s). Lower for low-FPS cameras.
gcode:
    # This block is for configuration storage only

# 2. THE "STORAGE" CONFIG
[save_variables]
filename: ~/printer_data/config/variables.cfg

# 3. THE LASER PIN CONFIG
[output_pin vdar_laser]
pin: mcu2:gpio18
pwm: true
value: 0
shutdown_value: 0

# 4. THE BASE SCAN ENGINE
[gcode_macro V_DAR_SCAN]
description: Base macro for V-DAR scanning
gcode:
    {% set conf = printer["gcode_macro V_DAR_CONFIG"] %}
    {% set MAT = params.MATERIAL|default("PLA") %}
    {% set E_TEMP = params.EXTRUDER_TEMP|default(200)|float %}
    {% set B_TEMP = params.BED_TEMP|default(60)|float %}
    {% set NOZZLE = params.NOZZLE_SIZE|default(conf.nozzle_size)|float %}
    {% set PA = params.PRESSURE_ADVANCE|default(0.0)|float %}
    {% set PA_ST = params.SMOOTH_TIME|default(0.04)|float %}

    M117 V-DAR: Scanning {MAT} ({NOZZLE}mm)
    G28
    G90
    
    # --- OPTIMIZE EXTRUSION ---
    SET_PRESSURE_ADVANCE ADVANCE={PA} SMOOTH_TIME={PA_ST}
    
    M117 V-DAR: Heating {MAT}
    M104 S{E_TEMP}
    M140 S{B_TEMP}
    M109 S{E_TEMP}
    M190 S{B_TEMP}

    # --- STEP 1: PRINT CALIBRATION PATTERN (RUBEDO STYLE) ---
    M117 V-DAR: Printing Pattern
    G1 Z0.3 F3000                # Go to print height
    M83                          # Relative extrusion
    
    # We print a series of lines with speed transitions to test flow/PA
    {% set start_x = 50 %}
    {% set end_x = 200 %}
    {% set start_y = 110 %}
    {% set line_spacing = 3 %}
    {% set num_lines = 10 %}
    
    {% for i in range(num_lines) %}
        {% set curr_y = start_y + (i * line_spacing) %}
        G1 X{start_x} Y{curr_y} F6000            # Move to start of line
        G1 X100 E{2.5 * (NOZZLE/0.4)} F600       # Slow (6mm/s)
        G1 X150 E{2.5 * (NOZZLE/0.4)} F9000      # Fast (150mm/s) - Pressure test
        G1 X200 E{2.5 * (NOZZLE/0.4)} F600       # Slow (6mm/s)
        G1 E-0.5 F1800                          # Small retract between lines
    {% endfor %}
    
    G1 E-0.5 F1800               # Final retract to prevent ooze
    
    # --- STEP 2: COOLDOWN (PREVENT CONTAMINATION) ---
    M117 V-DAR: Cooling Down...
    M104 S0                      # Turn off extruder heater
    M140 S0                      # Turn off bed heater
    G4 P2000                     # Wait 2 seconds for pressure to stabilize
    G1 Z5 F3000                  # Lift clear for laser

    # --- STEP 3: SEGMENTED LASER SCAN SWEEP (4 PARTS) ---
    M117 V-DAR: Scanning (Part 1/4)
    # Sweep 1: Lines 1-2 (Forward)
    G1 X125 Y{start_y - 3} F6000
    SET_PIN PIN=vdar_laser VALUE=1
    G1 Y{start_y + (1 * line_spacing) + 3} F{conf.scan_speed}
    SET_PIN PIN=vdar_laser VALUE=0

    M117 V-DAR: Scanning (Part 2/4)
    # Sweep 2: Lines 3-5 (Backward)
    G1 Y{start_y + (4 * line_spacing) + 3} F6000
    SET_PIN PIN=vdar_laser VALUE=1
    G1 Y{start_y + (2 * line_spacing) - 3} F{conf.scan_speed}
    SET_PIN PIN=vdar_laser VALUE=0

    M117 V-DAR: Scanning (Part 3/4)
    # Sweep 3: Lines 6-7 (Forward)
    G1 Y{start_y + (5 * line_spacing) - 3} F6000
    SET_PIN PIN=vdar_laser VALUE=1
    G1 Y{start_y + (6 * line_spacing) + 3} F{conf.scan_speed}
    SET_PIN PIN=vdar_laser VALUE=0

    M117 V-DAR: Scanning (Part 4/4)
    # Sweep 4: Lines 8-10 (Backward)
    G1 Y{start_y + (9 * line_spacing) + 3} F6000
    SET_PIN PIN=vdar_laser VALUE=1
    G1 Y{start_y + (7 * line_spacing) - 3} F{conf.scan_speed}
    SET_PIN PIN=vdar_laser VALUE=0
    G1 Z10 F3000                 # Lift clear
    
    M117 V-DAR: Processing...
    M118 V-DAR: Scan sweep finished. Service processing results...
    # NOTE: V_DAR_SET_OFFSET is called by the V-DAR service after analysis.
    
    M117 V-DAR: Scan Done

[gcode_macro V_DAR_PLA]
description: Quick PLA Scan
gcode:
    V_DAR_SCAN MATERIAL="PLA" EXTRUDER_TEMP=210 BED_TEMP=60 PRESSURE_ADVANCE=0.035

[gcode_macro V_DAR_ABS]

    V_DAR_SCAN MATERIAL="ABS" EXTRUDER_TEMP=250 BED_TEMP=110 PRESSURE_ADVANCE=0.045

description: Quick PETG Scan

    V_DAR_SCAN MATERIAL="PETG" EXTRUDER_TEMP=240 BED_TEMP=80 PRESSURE_ADVANCE=0.06

[gcode_macro V_DAR_RESET]
description: Resets all V-DAR offsets to 0
gcode:
    SAVE_VARIABLE VARIABLE=vdar_last_offset VALUE=0.0
    SET_GCODE_OFFSET Z=0 MOVE=1
    SET_PRESSURE_ADVANCE ADVANCE=0.0 SMOOTH_TIME=0.04
    M221 S100
    M117 V-DAR: Reset to Stock
    M118 V-DAR: All offsets cleared

# 5. INTEGRATION IN PRINT_START
# Use this in your printer.cfg to merge your heating logic with V-DAR
[gcode_macro PRINT_START]
gcode:
    # A. Load V-DAR Persistence
    {% set vdar_o = printer.save_variables.variables.vdar_last_offset|default(0.0) %}
    
    # B. Apply calculated offset
    SET_GCODE_OFFSET Z_ADJUST={vdar_o} MOVE=1
    M117 V-DAR: Applied {vdar_o} offset
    
    # C. Your original heating/homing logic
    M190 S50            # Heat bed
    M109 S150           # Heat hotend
    G28                 # Home all
    QUAD_GANTRY_LEVEL   # Level gantry
    G90                 # Absolute positioning
    G1 Z20 F3000        # Move clear
